Step 3: Register the City in the Database

# Run from backend/
.venv/bin/python -c "
from app.database import sync_session_factory
from sqlalchemy import text
import uuid

with sync_session_factory() as s:
    s.execute(text('''
        INSERT INTO cities (id, name, slug, state, center, bounds, default_zoom)
        VALUES (:id, :name, :slug, :state,
                ST_SetSRID(ST_MakePoint(:lng, :lat), 4326),
                ST_MakeEnvelope(:w, :s, :e, :n, 4326),
                :zoom)

        'id': uuid.uuid4(),

Step 5: Enrich Data (post-import)

After base data is imported, run these enrichment steps:

# 1. Spatial join: tag buildings with zoning codes
UPDATE buildings b SET zoning_code = z.zone_code, zoning_desc = z.zone_name
FROM zoning_districts z WHERE ST_Intersects(ST_Centroid(b.geom), z.geom)
AND b.zoning_code IS NULL AND b.city_id = 'YOUR_CITY_UUID';

# 2. Spatial join: link POIs to buildings
UPDATE pois SET building_id = b.id
FROM buildings b WHERE ST_Contains(b.geom, pois.geom)
AND pois.building_id IS NULL AND pois.city_id = 'YOUR_CITY_UUID';

# 3. Fetch business licenses (if your city has a Socrata portal)
# Download from city open data, insert into business_licenses table,
# then spatial match to buildings:
UPDATE buildings b SET owner_name = l.legal_name, metadata = jsonb_build_object('businesses', ...)
FROM business_licenses l WHERE ST_Contains(b.geom, l.geom)
AND b.zoning_code NOT LIKE 'R%';

# 4. Enrich addresses with full street names from building footprint data
# (same approach as Chicago — download source data with coordinates,
#  spatial match to buildings)

# 5. Import waterways from OSM

# ... or use Overpass for waterways

5. Frontend Layer Architecture

Layer Registry Pattern

Each map layer follows the same pattern. Files live in frontend/src/map/layers/:

// ExampleLayer.ts
import type maplibregl from 'maplibre-gl';
import type { LayerConfig } from '../../types/layer';

const SOURCE_ID = 'example-source';

export function addExampleLayer(map: maplibregl.Map, config: LayerConfig): void {
  if (map.getSource(SOURCE_ID)) return;  // idempotent

  map.addSource(SOURCE_ID, {
    type: 'vector',
    tiles: [config.sourceUrl],  // Martin tile URL
    minzoom: 0, maxzoom: 16,
  });

  map.addLayer({
    id: `${config.id}-fill`,
    type: 'fill',
    source: SOURCE_ID,
    'source-layer': 'TABLE_NAME',  // must match PostGIS table name
    layout: { visibility: config.visible ? 'visible' : 'none' },
    paint: { 'fill-color': '#color', 'fill-opacity': config.opacity * 0.3 },
  });
}

export function getExampleLayerIds(configId: string): string[] {
  return [`${configId}-fill`];
}

To add a new layer:

  1. Create the PostGIS table and import data
  2. Restart Martin (docker restart citymap-martin) — it auto-discovers new tables
  3. Create frontend/src/map/layers/YourLayer.ts following the pattern above
  4. Register in MapContainer.tsx LAYER_REGISTRY
  5. Add to useMapLayers.ts createDefaultLayers() and getLayerSourceUrl()
